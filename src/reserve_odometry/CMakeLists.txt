# A src-only workspace must carry all runtime assets. The root copy is retained
# for offline research scripts; full-repository builds detect accidental drift.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../../config)
  file(GLOB runtime_assets RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/config ${CMAKE_CURRENT_SOURCE_DIR}/config/*)
  foreach(asset ${runtime_assets})
    file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/config/${asset} packaged_sha)
    file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/../../config/${asset} research_sha)
    if(NOT packaged_sha STREQUAL research_sha)
      message(FATAL_ERROR "Runtime asset differs from research config: ${asset}")
    endif()
  endforeach()
endif()
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
